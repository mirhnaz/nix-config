# These are machine-wide CLIs, NOT per-project dependencies. Per-project Python

# `uv sync` there — they do NOT belong in this file.

# so `brew bundle install` will install them for you. This file is the
# brew-independent fallback — use it if you install uv tools without Homebrew.
#
# Regenerate:  uv tool list | awk '/^[^ -]/ {print $1}' > macos/uv-tools.txt
